package com.example.util

import android.content.Context
import java.security.MessageDigest
import java.util.UUID

class SquaresAdminManager(context: Context) {
    private val prefs = context.getSharedPreferences("squares_admin_prefs", Context.MODE_PRIVATE)

    fun isPinSet(): Boolean {
        return prefs.getString("pin_hash", null) != null
    }

    fun setPin(pin: String) {
        val salt = UUID.randomUUID().toString()
        val hash = hashPin(pin, salt)
        prefs.edit()
            .putString("pin_hash", hash)
            .putString("pin_salt", salt)
            .putInt("failed_attempts", 0)
            .putLong("lockout_until", 0L)
            .apply()
    }

    fun verifyPin(pin: String): Boolean {
        if (isLockedOut()) return false
        val hash = prefs.getString("pin_hash", null) ?: return false
        val salt = prefs.getString("pin_salt", null) ?: return false
        val matches = hashPin(pin, salt) == hash
        if (matches) {
            resetFailedAttempts()
        } else {
            recordFailedAttempt()
        }
        return matches
    }

    fun isLockedOut(): Boolean {
        val lockoutUntil = prefs.getLong("lockout_until", 0L)
        return System.currentTimeMillis() < lockoutUntil
    }

    fun getRemainingLockoutSeconds(): Int {
        val lockoutUntil = prefs.getLong("lockout_until", 0L)
        val now = System.currentTimeMillis()
        return if (lockoutUntil > now) ((lockoutUntil - now) / 1000).toInt() + 1 else 0
    }

    fun getFailedAttempts(): Int {
        return prefs.getInt("failed_attempts", 0)
    }

    private fun recordFailedAttempt(): Int {
        val current = prefs.getInt("failed_attempts", 0) + 1
        val editor = prefs.edit().putInt("failed_attempts", current)
        if (current >= 5) {
            editor.putLong("lockout_until", System.currentTimeMillis() + 60_000L)
        }
        editor.apply()
        return current
    }

    fun resetFailedAttempts() {
        prefs.edit()
            .putInt("failed_attempts", 0)
            .putLong("lockout_until", 0L)
            .apply()
    }

    private fun hashPin(pin: String, salt: String): String {
        val md = MessageDigest.getInstance("SHA-256")
        val bytes = md.digest((salt + pin).toByteArray(Charsets.UTF_8))
        return bytes.joinToString("") { "%02x".format(it) }
    }
}
